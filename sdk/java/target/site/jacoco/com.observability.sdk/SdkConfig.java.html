package com.observability.sdk;

import java.net.URI;
import java.nio.file.Path;
import java.util.Objects;
import java.util.Optional;

/**
 * Language-neutral SDK configuration model for the Observability SDK (Java).
 *
 * <p>Covers OTel Collector endpoint, TLS/cert-manager certificate paths,
 * and Vault authentication parameters. Instances are immutable and built
 * via {@link Builder}.
 *
 * <p>Configuration schema reference: {@code sdk/design/config-schema.json}
 */
public final class SdkConfig {

    /** OTLP transport protocol. */
    public enum Protocol {
        GRPC,
        HTTP_PROTOBUF,
        HTTP_JSON
    }

    /** Vault authentication method. */
    public enum VaultAuthMethod {
        APPROLE,
        KUBERNETES,
        TOKEN
    }

    /** SDK internal log level. */
    public enum LogLevel {
        DEBUG,
        INFO,
        WARN,
        ERROR
    }

    // ── Core fields ──────────────────────────────────────────────────
    private final URI endpoint;
    private final Protocol protocol;
    private final String serviceName;
    private final String environment;

    // ── TLS fields ───────────────────────────────────────────────────
    private final boolean tlsEnabled;
    private final Path tlsCertPath;
    private final Path tlsKeyPath;
    private final Path caPath;
    private final boolean insecureSkipVerify;
    private final String minTlsVersion;

    // ── Vault fields ─────────────────────────────────────────────────
    private final boolean vaultEnabled;
    private final URI vaultAddress;
    private final String vaultRole;
    private final VaultAuthMethod vaultAuthMethod;
    private final String vaultToken;
    private final String approleRoleId;
    private final String approleSecretId;
    private final String kubernetesServiceAccount;
    private final String secretPath;
    private final boolean autoRenew;

    // ── Batch fields ─────────────────────────────────────────────────
    private final int maxExportBatchSize;
    private final long scheduledDelayMs;
    private final long exportTimeoutMs;

    // ── Logging ──────────────────────────────────────────────────────
    private final LogLevel logLevel;

    private SdkConfig(Builder builder) {
        this.endpoint = Objects.requireNonNull(builder.endpoint, "endpoint must not be null");
        this.protocol = builder.protocol;
        this.serviceName = builder.serviceName;
        this.environment = builder.environment;
        this.tlsEnabled = builder.tlsEnabled;
        this.tlsCertPath = builder.tlsCertPath;
        this.tlsKeyPath = builder.tlsKeyPath;
        this.caPath = builder.caPath;
        this.insecureSkipVerify = builder.insecureSkipVerify;
        this.minTlsVersion = builder.minTlsVersion;
        this.vaultEnabled = builder.vaultEnabled;
        this.vaultAddress = builder.vaultAddress;
        this.vaultRole = builder.vaultRole;
        this.vaultAuthMethod = builder.vaultAuthMethod;
        this.vaultToken = builder.vaultToken;
        this.approleRoleId = builder.approleRoleId;
        this.approleSecretId = builder.approleSecretId;
        this.kubernetesServiceAccount = builder.kubernetesServiceAccount;
        this.secretPath = builder.secretPath;
        this.autoRenew = builder.autoRenew;
        this.maxExportBatchSize = builder.maxExportBatchSize;
        this.scheduledDelayMs = builder.scheduledDelayMs;
        this.exportTimeoutMs = builder.exportTimeoutMs;
        this.logLevel = builder.logLevel;
    }

    // ── Accessors ────────────────────────────────────────────────────

    public URI getEndpoint() { return endpoint; }
    public Protocol getProtocol() { return protocol; }
    public Optional<String> getServiceName() { return Optional.ofNullable(serviceName); }
    public Optional<String> getEnvironment() { return Optional.ofNullable(environment); }

    public boolean isTlsEnabled() { return tlsEnabled; }
    public Optional<Path> getTlsCertPath() { return Optional.ofNullable(tlsCertPath); }
    public Optional<Path> getTlsKeyPath() { return Optional.ofNullable(tlsKeyPath); }
    public Optional<Path> getCaPath() { return Optional.ofNullable(caPath); }
    public boolean isInsecureSkipVerify() { return insecureSkipVerify; }
    public String getMinTlsVersion() { return minTlsVersion; }

    public boolean isVaultEnabled() { return vaultEnabled; }
    public Optional<URI> getVaultAddress() { return Optional.ofNullable(vaultAddress); }
    public Optional<String> getVaultRole() { return Optional.ofNullable(vaultRole); }
    public VaultAuthMethod getVaultAuthMethod() { return vaultAuthMethod; }
    public Optional<String> getVaultToken() { return Optional.ofNullable(vaultToken); }
    public Optional<String> getApproleRoleId() { return Optional.ofNullable(approleRoleId); }
    public Optional<String> getApproleSecretId() { return Optional.ofNullable(approleSecretId); }
    public String getKubernetesServiceAccount() { return kubernetesServiceAccount; }
    public Optional<String> getSecretPath() { return Optional.ofNullable(secretPath); }
    public boolean isAutoRenew() { return autoRenew; }

    public int getMaxExportBatchSize() { return maxExportBatchSize; }
    public long getScheduledDelayMs() { return scheduledDelayMs; }
    public long getExportTimeoutMs() { return exportTimeoutMs; }
    public LogLevel getLogLevel() { return logLevel; }

     * Check the health of the SDK connection to the OTel Collector.
     *
     * @param sdk the SDK instance to check
     * @return health status
     */
    public static HealthStatus healthCheck(ObservabilitySdk sdk) {
        Objects.requireNonNull(sdk, "sdk must not be null");
        // Verify connectivity to OTel Collector endpoint
        throw new UnsupportedOperationException("SDK implementation not yet provided");
    }

    /** Health status returned by {@link #healthCheck}. */
    public enum HealthStatus {
        HEALTHY,
        DEGRADED,
        UNHEALTHY
    }

    /** Marker interface for the initialized SDK instance. */
    public interface ObservabilitySdk {
        /** Returns the configuration this SDK was initialized with. */
        SdkConfig getConfig();
    }

    // ── Builder ──────────────────────────────────────────────────────

    public static Builder builder(URI endpoint) {
        return new Builder(endpoint);
    }

    public static final class Builder {
        private final URI endpoint;
        private Protocol protocol = Protocol.HTTP_PROTOBUF;
        private String serviceName;
        private String environment;
        private boolean tlsEnabled = false;
        private Path tlsCertPath;
        private Path tlsKeyPath;
        private Path caPath;
        private boolean insecureSkipVerify = false;
        private String minTlsVersion = "1.2";
        private boolean vaultEnabled = false;
        private URI vaultAddress;
        private String vaultRole;
        private VaultAuthMethod vaultAuthMethod = VaultAuthMethod.APPROLE;
        private String vaultToken;
        private String approleRoleId;
        private String approleSecretId;
        private String kubernetesServiceAccount = "default";
        private String secretPath;
        private boolean autoRenew = true;
        private int maxExportBatchSize = 512;
        private long scheduledDelayMs = 5000;
        private long exportTimeoutMs = 30000;
        private LogLevel logLevel = LogLevel.INFO;

        private Builder(URI endpoint) {
            this.endpoint = Objects.requireNonNull(endpoint);
        }

        public Builder protocol(Protocol protocol) { this.protocol = protocol; return this; }
        public Builder serviceName(String serviceName) { this.serviceName = serviceName; return this; }
        public Builder environment(String environment) { this.environment = environment; return this; }

        public Builder tlsEnabled(boolean enabled) { this.tlsEnabled = enabled; return this; }
        public Builder tlsCertPath(Path path) { this.tlsCertPath = path; return this; }
        public Builder tlsKeyPath(Path path) { this.tlsKeyPath = path; return this; }
        public Builder caPath(Path path) { this.caPath = path; return this; }
        public Builder insecureSkipVerify(boolean skip) { this.insecureSkipVerify = skip; return this; }
        public Builder minTlsVersion(String version) { this.minTlsVersion = version; return this; }

        public Builder vaultEnabled(boolean enabled) { this.vaultEnabled = enabled; return this; }
        public Builder vaultAddress(URI address) { this.vaultAddress = address; return this; }
        public Builder vaultRole(String role) { this.vaultRole = role; return this; }
        public Builder vaultAuthMethod(VaultAuthMethod method) { this.vaultAuthMethod = method; return this; }
        public Builder vaultToken(String token) { this.vaultToken = token; return this; }
        public Builder approleRoleId(String roleId) { this.approleRoleId = roleId; return this; }
        public Builder approleSecretId(String secretId) { this.approleSecretId = secretId; return this; }
        public Builder kubernetesServiceAccount(String sa) { this.kubernetesServiceAccount = sa; return this; }
        public Builder secretPath(String path) { this.secretPath = path; return this; }
        public Builder autoRenew(boolean renew) { this.autoRenew = renew; return this; }

        public Builder maxExportBatchSize(int size) { this.maxExportBatchSize = size; return this; }
        public Builder scheduledDelayMs(long ms) { this.scheduledDelayMs = ms; return this; }
        public Builder exportTimeoutMs(long ms) { this.exportTimeoutMs = ms; return this; }
        public Builder logLevel(LogLevel level) { this.logLevel = level; return this; }

        public SdkConfig build() {
            return new SdkConfig(this);
        }
    }
}
