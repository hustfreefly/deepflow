package com.observability.sdk;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Objects;
import java.util.Optional;

/**
 * Resolves a Vault token from the configured source.
 *
 * <p>Token resolution order (first non-empty wins):
 * <ol>
 *   <li>Environment variable {@code VAULT_TOKEN}</li>
 *   <li>File at {@code /var/run/secrets/vault/token} (Kubernetes CSI driver)</li>
 *   <li>Direct configuration value ({@code vault.token} in config)</li>
 * </ol>
 *
 * <p>The resolved token is intended to be injected as an {@code X-Vault-Token}
 * header on the OTLP exporter so the collector can authenticate with Vault.
 *
 * <p>Usage:
 * <pre>{@code
 *   VaultTokenProvider provider = new VaultTokenProvider(config);
 *   Optional<String> token = provider.resolveToken();
 *   token.ifPresent(t -> exporterBuilder.addHeader("X-Vault-Token", t));
 * }</pre>
 */
public final class VaultTokenProvider {

    /** Environment variable name for Vault token. */
    public static final String VAULT_TOKEN_ENV = "VAULT_TOKEN";

    /** Default file path for Vault token (Kubernetes CSI driver mount). */
    public static final String DEFAULT_VAULT_TOKEN_FILE = "/var/run/secrets/vault/token";

    private final SdkConfig config;

    /**
     * Creates a provider for the given SDK configuration.
     *
     * @param config the SDK configuration (must not be null)
     */
    public VaultTokenProvider(SdkConfig config) {
        this.config = Objects.requireNonNull(config, "config must not be null");
    }

    /**
     * Returns true if Vault integration is enabled in the configuration.
     */
    public boolean isVaultEnabled() {
        return config.isVaultEnabled();
    }

    /**
     * Resolves the Vault token using the configured resolution order.
     *
     * <p>Resolution order: env var → file → direct config value.
     * Returns empty if no token is found.
     *
     * @return the resolved token, or empty
     */
    public Optional<String> resolveToken() {
        if (!config.isVaultEnabled()) {
            return Optional.empty();
        }

        // 1. Environment variable
        String envToken = resolveFromEnv(VAULT_TOKEN_ENV);
        if (envToken != null && !envToken.isEmpty()) {
            return Optional.of(envToken);
        }

        // 2. Token file (Kubernetes CSI driver)
        String fileToken = resolveFromFile(DEFAULT_VAULT_TOKEN_FILE);
        if (fileToken != null && !fileToken.isEmpty()) {
            return Optional.of(fileToken);
        }

        // 3. Direct config value
        return config.getVaultToken();
    }

    /**
     * Resolves token from an environment variable.
     *
     * @param envVar the environment variable name
     * @return the value, or null if not set
     */
    public String resolveFromEnv(String envVar) {
        return System.getenv(envVar);
    }

    /**
     * Resolves token from a file.
     *
     * @param filePath the path to the token file
     * @return the trimmed file content, or null if the file does not exist
     */
    public String resolveFromFile(String filePath) {
        Path path = Paths.get(filePath);
        if (!Files.exists(path) || !Files.isReadable(path)) {
            return null;
        }
        try {
            return new String(Files.readAllBytes(path)).trim();
        } catch (IOException e) {
            return null;
        }
    }

    /**
     * Returns the Vault address from the configuration.
     *
     * @return the Vault address, or empty if not configured
     */
    public Optional<String> getVaultAddress() {
        return config.getVaultAddress().map(Object::toString);
    }
}
