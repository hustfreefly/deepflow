package com.observability.sdk;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Objects;
import java.util.Optional;

/**
 * Loads TLS certificate material from filesystem paths.
 *
 * <p>Certificates are expected to be mounted by cert-manager at the configured
 * paths. This loader reads PEM-encoded certificate and key files into byte arrays
 * suitable for the OpenTelemetry OTLP exporter builder.
 *
 * <p>Usage:
 * <pre>{@code
 *   TlsConfigLoader loader = new TlsConfigLoader(config);
 *   byte[] trustedCerts = loader.loadTrustedCertificates();  // CA cert
 *   byte[] clientCert = loader.loadClientCertificateChain();  // client cert
 *   byte[] clientKey = loader.loadClientKey();                // client private key
 * }</pre>
 */
public final class TlsConfigLoader {

    private final SdkConfig config;

    /**
     * Creates a loader for the given SDK configuration.
     *
     * @param config the SDK configuration (must not be null)
     */
    public TlsConfigLoader(SdkConfig config) {
        this.config = Objects.requireNonNull(config, "config must not be null");
    }

    /**
     * Returns true if TLS is enabled in the configuration.
     */
    public boolean isTlsEnabled() {
        return config.isTlsEnabled();
    }

    /**
     * Loads the trusted CA certificate bundle.
     *
     * @return the CA certificate bytes, or empty if no CA path is configured
     * @throws IOException if the file cannot be read
     */
    public Optional<byte[]> loadTrustedCertificates() throws IOException {
        return config.getCaPath().map(this::readAllBytes);
    }

    /**
     * Loads the client certificate chain (for mTLS).
     *
     * @return the client certificate bytes, or empty if no cert path is configured
     * @throws IOException if the file cannot be read
     */
    public Optional<byte[]> loadClientCertificateChain() throws IOException {
        return config.getTlsCertPath().map(this::readAllBytes);
    }

    /**
     * Loads the client private key (for mTLS).
     *
     * @return the client key bytes, or empty if no key path is configured
     * @throws IOException if the file cannot be read
     */
    public Optional<byte[]> loadClientKey() throws IOException {
        return config.getTlsKeyPath().map(this::readAllBytes);
    }

    /**
     * Validates that all configured TLS paths exist and are readable.
     *
     * @throws IOException if any configured path is missing or unreadable
     */
    public void validateAllPaths() throws IOException {
        if (!config.isTlsEnabled()) {
            return;
        }

        config.getCaPath().ifPresent(this::validatePath);
        config.getTlsCertPath().ifPresent(this::validatePath);
        config.getTlsKeyPath().ifPresent(this::validatePath);
    }

    private void validatePath(Path path) {
        if (!Files.exists(path)) {
            throw new IllegalStateException(
                    "TLS file not found: " + path.toAbsolutePath());
        }
        if (!Files.isReadable(path)) {
            throw new IllegalStateException(
                    "TLS file not readable: " + path.toAbsolutePath());
        }
    }

    private byte[] readAllBytes(Path path) {
        try {
            return Files.readAllBytes(path);
        } catch (IOException e) {
            throw new IllegalStateException(
                    "Failed to read TLS file: " + path.toAbsolutePath(), e);
        }
    }
}
